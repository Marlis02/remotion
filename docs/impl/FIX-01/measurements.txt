  // ЗОНД №182: где именно теряется кадр при `scale < 1`.
  // Каталог `out/` — в `.gitignore` и в `ignores` eslint; в дифф не входит.
  import { renderSegment } from '../dist/src/run.js';
  import { decodeRgb, pngSize } from '../dist/src/where.js';
  import { makeTemplateFixture, readyRequest } from '../dist/test/fixture.js';
  import { TEST_REGISTRY } from '../dist/test/solid.js';
  import path from 'node:path';
  
  const clock = () => { let t = 0; return () => (t += 10); };
  
  async function shot(scale) {
    const fixture = makeTemplateFixture(
      [{ template: 'solid@1', params: { color: '#204080' }, z: 0 }],
      { frames: 2, scale },
    );
    const request = await readyRequest(fixture.request, TEST_REGISTRY);
    const r = await renderSegment(request, {
      clock: clock(), gate: { mode: 'skip', why: 'зонд №182' },
      registry: TEST_REGISTRY, parentEnv: process.env,
    });
    if (!r.ok) throw new Error(`${r.error.rule}: ${r.error.message}`);
    const file = path.join(r.frames.dir, 'frame_000001.png');
    const size = pngSize(file);
    const buf = await decodeRgb(file);
    // Доля НЕЧЁРНЫХ пикселей по четырём квадрантам.
    const q = [0, 0, 0, 0], n = [0, 0, 0, 0];
    for (let y = 0; y < size.height; y++) {
      for (let x = 0; x < size.width; x++) {
        const i = (y * size.width + x) * 3;
        const lit = buf[i] !== 0 || buf[i + 1] !== 0 || buf[i + 2] !== 0;
        const k = (y < size.height / 2 ? 0 : 2) + (x < size.width / 2 ? 0 : 1);
        n[k]++; if (lit) q[k]++;
      }
    }
    const pct = q.map((v, k) => Math.round((v / n[k]) * 1000) / 10);
    const px = (yy, xx) => { const i = (yy * size.width + xx) * 3; return `${buf[i]},${buf[i+1]},${buf[i+2]}`; };
    console.log(`scale ${scale}: PNG ${size.width}x${size.height}`);
    console.log(`  нечёрных по квадрантам, %: ЛВ ${pct[0]}  ПВ ${pct[1]}  ЛН ${pct[2]}  ПН ${pct[3]}`);
    console.log(`  проба пикселей: центр ЛВ=${px(size.height >> 2, size.width >> 2)}  центр ПН=${px((size.height * 3) >> 2, (size.width * 3) >> 2)}`);
    return { size, pct };
  }
  
  for (const s of [1, 0.5]) await shot(s);

═══ ЗОНД 3: probe-easing-form.mjs — форма «похожести» кривой (§4 отчёта) ═══
  // Замер формы «похожести» для №184: сколько литералов РАЗРЕШЁННОГО `mountSource` она ловит
  // и сколько из них — ложные (не кривые).
  import { rendererTemplates } from '../dist/src/templates/index.js';
  import { SOLID_TEMPLATE } from '../dist/test/solid.js';
  import { EASING_REGISTRY } from '@vpe/templates-spec';
  
  const LOOKS = /^[A-Za-z][A-Za-z0-9]*\.[A-Za-z][A-Za-z0-9]*(\([^)]*\))?$|^none$/u;
  const literals = (src) => [...src.matchAll(/'([^'\\]*)'|"([^"\\]*)"/gu)].map((m) => m[1] ?? m[2] ?? '');
  
  let total = 0, hits = 0;
  for (const impl of [...rendererTemplates.templates, SOLID_TEMPLATE]) {
    const all = literals(impl.mountSource);
    const matched = all.filter((s) => LOOKS.test(s));
    total += all.length; hits += matched.length;
    console.log(`${impl.templateId}@${impl.templateVersion}: литералов ${all.length}, похожих на кривую ${matched.length}` +
      (matched.length ? ` → ${matched.map((s) => `'${s}'` + (EASING_REGISTRY.includes(s) ? ' [в реестре]' : ' [ВНЕ РЕЕСТРА]')).join(', ')}` : ''));
  }
  console.log(`ИТОГО: литералов ${total}, срабатываний ${hits}`);
  console.log(`проба Н1в: 'elastic.out' → ловится=${LOOKS.test('elastic.out')}, в реестре=${EASING_REGISTRY.includes('elastic.out')}`);
  console.log(`контроль форм: 'back.out(1.7)'→${LOOKS.test('back.out(1.7)')}, 'none'→${LOOKS.test('none')}, 'cover'→${LOOKS.test('cover')}, '#ffffff'→${LOOKS.test('#ffffff')}, '50%'→${LOOKS.test('50%')}, 'style.transform'→${LOOKS.test('style.transform')}`);
